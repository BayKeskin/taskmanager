async function runAiBriefDecomposition() {
    const brief = document.getElementById('aiArchBriefInput').value.trim();
    const projId = document.getElementById('aiArchProjectSelect').value;
    if (!brief) {
        showToast('Lütfen önce proje brifini veya özellik listesini yazın.', 'error');
        return;
    }

    const btn = document.getElementById('btnRunAiDecompose');
    const origText = btn.innerHTML;
    btn.disabled = true;
    btn.innerHTML = '⏳ AI Mimarı Görev Ağacını Kuruyor...';

    try {
        const fd = new FormData();
        fd.append('brief', brief);
        fd.append('project_id', projId);

        const res = await fetch('/taskmanager/ajax?action=decompose_brief_with_ai', { method: 'POST', body: fd });
        const data = await res.json();

        if (data.success && data.epics) {
            aiArchTreeData = data.epics.map(ep => ({
                title: ep.title || 'Modül',
                description: ep.description || '',
                priority: ep.priority || 'high',
                execution_mode: ep.execution_mode === 'gated' ? 'gated' : 'auto',
                subtasks: (ep.subtasks || []).map(st => ({
                    title: st.title || '',
                    description: st.description || '',
                    category: st.category || 'coding',
                    priority: st.priority || 'normal',
                    time_estimate: st.time_estimate || '2h',
                    assigned_to: (window.CURRENT_USER_ID || 1),
                    is_ai: false // Varsayılan olarak ASLA otomatik AI seçili gelmez!
                }))
            }));

            document.getElementById('aiArchStep1').style.display = 'none';

        }
    } catch (e) {
        showToast('Sunucu bağlantı hatası', 'error');
    } finally {
        btn.disabled = false;
        btn.innerHTML = origText;

function renderAiArchitectTree() {
    const container = document.getElementById('aiArchEpicsContainer');
    if (!container) return;
    container.innerHTML = '';

    const teamUsers = window.ALL_TEAM_USERS || [{ id: 1, name: 'Yönetici' }];
    let totalSubs = 0;
    let totalAiSubs = 0;

    aiArchTreeData.forEach((epic, eIdx) => {
        const epicBox = document.createElement('div');
        epicBox.style.cssText = 'background:var(--bg-card); border:1px solid var(--border-subtle); border-radius:10px; padding:14px; transition:all 0.2s;';
        
        // Dropzone Events (Alt görevi başka bir ana göreve sürükleyip bırakma)
        epicBox.ondragover = (ev) => {
            ev.preventDefault();
            epicBox.style.borderColor = '#7B68EE';
            epicBox.style.boxShadow = '0 0 0 2px rgba(123,104,238,0.25)';
        };
        epicBox.ondragleave = () => {
            epicBox.style.borderColor = 'var(--border-subtle)';
            epicBox.style.boxShadow = 'none';
        };
        epicBox.ondrop = (ev) => {
            ev.preventDefault();
            epicBox.style.borderColor = 'var(--border-subtle)';
            epicBox.style.boxShadow = 'none';
            if (!draggedSubtaskRef) return;
            const { epicIdx: fromE, subIdx: fromS } = draggedSubtaskRef;
            if (fromE === eIdx) return;

            const movedItem = aiArchTreeData[fromE].subtasks.splice(fromS, 1)[0];
            if (movedItem) {
                aiArchTreeData[eIdx].subtasks.push(movedItem);
                draggedSubtaskRef = null;
                renderAiArchitectTree();
                showToast('Alt görev yeni modüle taşındı!', 'info');
            }
        };

        const isGated = (epic.execution_mode === 'gated');

        let subtasksHtml = '';
        (epic.subtasks || []).forEach((sub, sIdx) => {
            totalSubs++;
            if (sub.is_ai) totalAiSubs++;

            const userOptions = teamUsers.map(u => `<option value="${u.id}" ${sub.assigned_to == u.id ? 'selected' : ''}>👤 ${escapeHtml(u.name)}</option>`).join('');

            subtasksHtml += `
                <div draggable="true"
                     ondragstart="draggedSubtaskRef = { epicIdx: ${eIdx}, subIdx: ${sIdx} }; this.style.opacity='0.4';"
                     ondragend="this.style.opacity='1'; draggedSubtaskRef = null;"
                     style="display:grid; grid-template-columns: 24px 1fr 110px 140px 95px 30px; gap:8px; align-items:center; background:var(--bg-base); border:1px solid ${sub.is_ai ? 'rgba(224,86,253,0.5)' : 'var(--border-subtle)'}; border-radius:6px; padding:8px 10px; margin-bottom:6px; cursor:grab;">
                    <span style="color:var(--text-muted); font-size:15px; user-select:none;" title="Sürükle ve başka modüle bırak">⠿</span>
                    <input type="text" value="${escapeHtml(sub.title)}" oninput="aiArchTreeData[${eIdx}].subtasks[${sIdx}].title = this.value" style="background:transparent; border:none; color:var(--text-white); font-size:12.5px; font-weight:500; outline:none; width:100%;">
                    <select onchange="aiArchTreeData[${eIdx}].subtasks[${sIdx}].category = this.value" style="background:var(--bg-card); border:1px solid var(--border-subtle); color:var(--text-secondary); font-size:11px; padding:4px 6px; border-radius:4px;">
                        <option value="coding" ${sub.category === 'coding' ? 'selected' : ''}>💻 Yazılım</option>
                        <option value="design" ${sub.category === 'design' ? 'selected' : ''}>🎨 Tasarım</option>
                        <option value="seo" ${sub.category === 'seo' ? 'selected' : ''}>🔍 SEO</option>
                        <option value="marketing" ${sub.category === 'marketing' ? 'selected' : ''}>📢 Pazarlama</option>
                        <option value="legal" ${sub.category === 'legal' ? 'selected' : ''}>⚖️ Yasal/KVKK</option>
                        <option value="content" ${sub.category === 'content' ? 'selected' : ''}>📝 İçerik</option>
                    </select>
                    <select onchange="aiArchTreeData[${eIdx}].subtasks[${sIdx}].assigned_to = parseInt(this.value)" style="background:var(--bg-card); border:1px solid var(--border-subtle); color:var(--text-secondary); font-size:11px; padding:4px 6px; border-radius:4px;">
                        ${userOptions}
                    </select>
                    <label style="display:inline-flex; align-items:center; justify-content:center; gap:5px; font-size:11.5px; font-weight:700; padding:4px 8px; border-radius:5px; cursor:pointer; background:${sub.is_ai ? 'linear-gradient(135deg, #7B68EE, #e056fd)' : 'rgba(255,255,255,0.05)'}; color:${sub.is_ai ? '#fff' : 'var(--text-muted)'}; border:1px solid ${sub.is_ai ? '#e056fd' : 'var(--border-subtle)'};">
                        <input type="checkbox" ${sub.is_ai ? 'checked' : ''} onchange="aiArchTreeData[${eIdx}].subtasks[${sIdx}].is_ai = this.checked; renderAiArchitectTree();" style="display:none;">
                        <span>🤖 AI</span>
                    </label>
                    <button type="button" onclick="aiArchTreeData[${eIdx}].subtasks.splice(${sIdx}, 1); renderAiArchitectTree();" style="background:none; border:none; color:#ff4757; cursor:pointer; font-size:13px;" title="Alt Görevi Sil">🗑️</button>
                </div>
            `;
        });

        epicBox.innerHTML = `
            <div style="display:flex; justify-content:space-between; align-items:center; gap:10px; margin-bottom:10px; padding-bottom:8px; border-bottom:1px solid var(--border-subtle);">
                <div style="display:flex; align-items:center; gap:8px; flex:1;">
                    <span style="font-size:16px;">📂</span>
                    <input type="text" value="${escapeHtml(epic.title)}" oninput="aiArchTreeData[${eIdx}].title = this.value" style="background:transparent; border:1px dashed transparent; color:var(--text-white); font-size:14px; font-weight:800; width:100%; padding:2px 6px; border-radius:4px;" onfocus="this.style.borderColor='var(--brand-primary)'" onblur="this.style.borderColor='transparent'">
                </div>
                <div style="display:flex; align-items:center; gap:8px;">
                    <select onchange="aiArchTreeData[${eIdx}].execution_mode = this.value; renderAiArchitectTree();" style="background:${isGated ? 'rgba(230,126,34,0.18)' : 'rgba(0,184,148,0.18)'}; border:1px solid ${isGated ? '#e67e22' : '#00b894'}; color:${isGated ? '#e67e22' : '#00b894'}; font-size:11px; font-weight:700; padding:4px 8px; border-radius:6px; cursor:pointer;">
                        <option value="auto" ${!isGated ? 'selected' : ''}>🚀 Tam Otonom Akış (Gece Modu)</option>
                        <option value="gated" ${isGated ? 'selected' : ''}>🛑 Onay Kapılı (Adım Adım Onaylı)</option>
                    </select>
                    <button type="button" onclick="addSubtaskToEpic(${eIdx})" style="background:rgba(123,104,238,0.15); border:1px solid rgba(123,104,238,0.4); color:#d8b4fe; font-size:11px; font-weight:700; padding:4px 8px; border-radius:5px; cursor:pointer;">+ Alt Görev</button>
                    <button type="button" onclick="aiArchTreeData.splice(${eIdx}, 1); renderAiArchitectTree();" style="background:rgba(255,71,87,0.12); border:1px solid rgba(255,71,87,0.3); color:#ff4757; font-size:11px; padding:4px 8px; border-radius:5px; cursor:pointer;" title="Modülü Sil">✕</button>
                </div>
            </div>
            <div>${subtasksHtml || '<div style="font-size:12px; color:var(--text-muted); padding:8px; text-align:center; border:1px dashed var(--border-subtle); border-radius:6px;">Bu modülde alt görev yok. Buraya alt görev sürükleyebilir veya "+ Alt Görev" butonuna basabilirsiniz.</div>'}</div>
        `;
        container.appendChild(epicBox);
    });

    document.getElementById('aiArchSummaryBadge').innerHTML = `📊 Toplam <b>${aiArchTreeData.length} Ana Modül</b>, <b>${totalSubs} Alt Görev</b> (<b>${totalAiSubs} tanesi 🤖 AI Otonom</b> olarak işaretlendi)`;
}

function addNewBlankEpic() {
    aiArchTreeData.push({
        title: 'Yeni Modül / Ana Görev',
        description: '',
        priority: 'normal',
        execution_mode: 'auto',
        subtasks: [
            { title: 'Yeni alt görev 1', description: '', category: 'coding', priority: 'normal', time_estimate: '2h', assigned_to: (window.CURRENT_USER_ID || 1), is_ai: false }
        ]
    });
    renderAiArchitectTree();
}

function addSubtaskToEpic(epicIdx) {
    aiArchTreeData[epicIdx].subtasks.push({
        title: 'Yeni alt görev adımı',
        description: '',
        category: 'coding',
        priority: 'normal',
        time_estimate: '2h',
        assigned_to: (window.CURRENT_USER_ID || 1),
        is_ai: false
    });
    renderAiArchitectTree();
}

async function submitAiArchitectTree() {
    if (aiArchTreeData.length === 0) {
        showToast('Kaydedilecek modül bulunamadı.', 'error');
        return;
    }

    const projId = document.getElementById('aiArchProjectSelect').value;
    const btn = document.getElementById('btnApproveAiArch');
    const origText = btn.innerHTML;
    btn.disabled = true;
    btn.innerHTML = '⏳ Projeye İşleniyor...';

    try {
        const fd = new FormData();
        fd.append('project_id', projId);
        fd.append('tree_json', JSON.stringify(aiArchTreeData));

        const res = await fetch('/taskmanager/ajax?action=bulk_create_hierarchical_tasks', { method: 'POST', body: fd });
        const data = await res.json();

        if (data.success) {
            showToast(data.message, 'success');
            closeModal('modalAiArchitect');
            setTimeout(() => window.location.reload(), 800);
        } else {
            showToast(data.message || 'Kaydedilemedi', 'error');
        }
    } catch (e) {
        showToast('Sunucu bağlantı hatası', 'error');
    } finally {
        btn.disabled = false;
        btn.innerHTML = origText;
    }
}
