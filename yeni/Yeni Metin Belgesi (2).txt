    /**
     * Yüklenen Dökümandan (.docx, .pdf, .txt, .md, görsel) Metin ve Gömülü Resimleri Çıkarır.
     * Word (.docx) içindeki 'word/media/*' resimlerini otomatik olarak Public/uploads/tasks/ altına kaydeder!
     */
    public static function extractContentAndImagesFromDocument(array $file): array
    {
        $result = [
            'text'           => '',
            'extracted_imgs' => [], // [['file_name' => ..., 'file_path' => ..., 'file_type' => ..., 'file_size' => ...]]
            'original_doc'   => null
        ];

        if (empty($file['tmp_name']) || !is_uploaded_file($file['tmp_name'])) {
            return $result;
        }

        $origName = $file['name'] ?? 'document';
        $ext = strtolower(pathinfo($origName, PATHINFO_EXTENSION));
        $uploadDir = dirname(__DIR__) . '/Public/uploads/tasks';
        if (!is_dir($uploadDir)) {
            @mkdir($uploadDir, 0777, true);
        }

        // 1. Orijinal dosyayı da görevlere bağlamak üzere kaydet
        $safeOrigName = 'brief_' . time() . '_' . preg_replace('/[^a-zA-Z0-9\._-]/', '_', $origName);
        $origDestPath = $uploadDir . '/' . $safeOrigName;
        if (@copy($file['tmp_name'], $origDestPath)) {
            $result['original_doc'] = [
                'file_name' => $origName,
                'file_path' => '/taskmanager/Public/uploads/tasks/' . $safeOrigName,
                'file_type' => $file['type'] ?? 'application/octet-stream',
                'file_size' => (int)($file['size'] ?? filesize($origDestPath))
            ];
        }

        // 2. Doğrudan Görsel Yüklendiyse (PNG, JPG, WEBP)
        if (in_array($ext, ['png', 'jpg', 'jpeg', 'webp', 'gif'], true)) {
            if ($result['original_doc']) {
                $result['extracted_imgs'][] = $result['original_doc'];
            }
            return $result;
        }

        // 3. Düz MetinDosyaları (.txt, .md, .json, .csv, .html)
        if (in_array($ext, ['txt', 'md', 'markdown', 'csv', 'json', 'html', 'htm'], true)) {
            $raw = (string)@file_get_contents($file['tmp_name']);
            $result['text'] = strip_tags($raw);
            return $result;
        }

        // 4. WORD (.DOCX) DOSYASI: Metinleri (Paragraflar + Tablolar) ve Gömülü Resimleri (word/media/*) Çıkar!
        if ($ext === 'docx' && class_exists('ZipArchive')) {
            $zip = new \ZipArchive();
            if ($zip->open($file['tmp_name']) === true) {
                // A) Metinleri Çıkar (word/document.xml)
                $xmlIndex = $zip->locateName('word/document.xml');
                if ($xmlIndex !== false) {
                    $xmlData = $zip->getFromIndex($xmlIndex);
                    // Paragraf ve tablo satır sonlarını koru
                    $xmlData = str_replace(['</w:p>', '</w:tr>', '<w:br/>', '<w:cr/>'], ["\n", "\n", "\n", "\n"], $xmlData);
                    $xmlData = str_replace(['</w:tc>'], ["\t"], $xmlData);
                    $cleanText = html_entity_decode(strip_tags($xmlData), ENT_QUOTES | ENT_XML1, 'UTF-8');
                    $result['text'] = trim(preg_replace("/\n{3,}/", "\n\n", $cleanText));
                }

                // B) Word İçindeki Resimleri Çıkar (word/media/*)
                for ($i = 0; $i < $zip->numFiles; $i++) {
                    $entryName = $zip->getNameIndex($i);
                    if (str_starts_with($entryName, 'word/media/')) {
                        $imgExt = strtolower(pathinfo($entryName, PATHINFO_EXTENSION));
                        if (in_array($imgExt, ['png', 'jpg', 'jpeg', 'gif', 'webp', 'bmp'], true)) {
                            $imgBuffer = $zip->getFromIndex($i);
                            if ($imgBuffer && strlen($imgBuffer) > 512) { // Çok küçük ikon/çizgileri atla (>512 byte)
                                $savedImgName = 'docx_img_' . time() . '_' . ($i + 1) . '.' . $imgExt;
                                $savedFullPath = $uploadDir . '/' . $savedImgName;
                                if (@file_put_contents($savedFullPath, $imgBuffer)) {
                                    $result['extracted_imgs'][] = [
                                        'file_name' => basename($entryName),
                                        'file_path' => '/taskmanager/Public/uploads/tasks/' . $savedImgName,
                                        'file_type' => 'image/' . ($imgExt === 'jpg' ? 'jpeg' : $imgExt),
                                        'file_size' => strlen($imgBuffer)
                                    ];
                                }
                            }
                        }
                    }
                }
                $zip->close();
            }
            return $result;
        }

        // 5. PDF Dosyası (Basit Metin Çıkarma)
        if ($ext === 'pdf') {
            $rawPdf = (string)@file_get_contents($file['tmp_name']);
            $extracted = '';
            if (preg_match_all('/\(([^\)]+)\)\s*Tj/s', $rawPdf, $matches)) {
                $extracted = implode(' ', $matches[1]);
            }
            if (mb_strlen(trim($extracted)) < 20 && preg_match_all('/\[([^\]]+)\]\s*TJ/s', $rawPdf, $matches2)) {
                $extracted .= ' ' . preg_replace('/[\(\)\-\d\.]+/', ' ', implode(' ', $matches2[1]));
            }
            $result['text'] = trim($extracted);
        }

        return $result;
    }

            if (preg_match('/(reklam|kampanya|pazarlama|sosyal medya|instagram|bülten|mail|bildirim|analitik)/u', $l)) return 'marketing';
            if (preg_match('/(sözleşme|kvkk|yasal|hukuk|gizlilik|iade|şartname|çerez|aydınlatma)/u', $l)) return 'legal';
            if (preg_match('/(yazı|makale|içerik|metin|blog|açıklama|çeviri|dil)/u', $l)) return 'content';

                        . "2. Sayfa veya form bağlantısını oluşturup kullanıcı erişimine aç.\n"
                        . "3. İçeriğin tüm cihazlarda düzgün görüntülendiğini doğrula.\n";

    dz.style.borderColor = 'var(--border-subtle)';
    dz.style.background = 'rgba(255,255,255,0.02)';
    if (e.dataTransfer.files && e.dataTransfer.files.length > 0) {
        setAiArchFile(e.dataTransfer.files[0]);
    }
}

async function runAiBriefDecomposition() {
    const brief = document.getElementById('aiArchBriefInput').value.trim();
    const projId = document.getElementById('aiArchProjectSelect').value;
    if (!brief && !selectedAiArchFile) {
        showToast('Lütfen proje brifini yazın veya bir Word (.docx) / döküman dosyası yükleyin.', 'error');
        return;
    }

    const btn = document.getElementById('btnRunAiDecompose');
    const origText = btn.innerHTML;
    btn.disabled = true;
    btn.innerHTML = '⏳ Döküman & Görev Adımları Analiz Ediliyor...';

    try {
        const fd = new FormData();
        fd.append('brief', brief);
        fd.append('project_id', projId);
        if (selectedAiArchFile) {
            fd.append('brief_file', selectedAiArchFile);
        }

        const res = await fetch('/taskmanager/ajax?action=decompose_brief_with_ai', { method: 'POST', body: fd });
        const data = await res.json();

        if (data.success && data.epics) {
            aiArchExtractedImgs = data.extracted_imgs || [];
            aiArchOriginalDoc = data.original_doc || null;

            aiArchTreeData = data.epics.map(ep => ({
                title: ep.title || 'Modül',
                description: ep.description || '',
                priority: ep.priority || 'high',
                execution_mode: ep.execution_mode === 'gated' ? 'gated' : 'auto',
                subtasks: (ep.subtasks || []).map(st => ({
                    title: st.title || '',
                    description: st.description || '🛠️ NASIL YAPILACAK:\n1. İlgili geliştirmeyi tamamla ve canlı ortamda test et.',
                    category: st.category || 'coding',
                    priority: st.priority || 'normal',
                    time_estimate: st.time_estimate || '2h',
                    assigned_to: (window.CURRENT_USER_ID || 1),
                    is_ai: false,
                    show_desc: true // Açıklamalar varsayılan olarak açık ve görünür gelir!
                }))
            }));

            // Çıkarılan resimler varsa önizleme barını doldur
            const imgBar = document.getElementById('aiArchExtractedImagesBar');
            const imgList = document.getElementById('aiArchExtractedImagesList');
            if (aiArchExtractedImgs.length > 0 && imgBar && imgList) {
                imgBar.style.display = 'block';
                imgList.innerHTML = aiArchExtractedImgs.map((img, idx) => `
                    <div style="position:relative; flex-shrink:0; width:90px; background:var(--bg-base); border:1px solid var(--border-subtle); border-radius:6px; overflow:hidden; text-align:center;">
                        <img src="${img.file_path}" style="width:90px; height:60px; object-fit:cover; display:block; cursor:pointer;" onclick="if(typeof openImageLightbox==='function') openImageLightbox('${img.file_path}', '${escapeHtml(img.file_name)}')">
                        <div style="font-size:9.5px; color:var(--text-muted); padding:2px 4px; white-space:nowrap; overflow:hidden; text-overflow:ellipsis;">${escapeHtml(img.file_name)}</div>
                    </div>
                `).join('');
            } else if (imgBar) {
                imgBar.style.display = 'none';
            }

            document.getElementById('aiArchStep1').style.display = 'none';
            document.getElementById('aiArchStep2').style.display = 'block';
            document.getElementById('btnApproveAiArch').style.display = 'inline-block';
            renderAiArchitectTree();
            showToast('✨ Görev ağacı, uygulama adımları ve döküman görselleri hazır!', 'success');
        } else {
            showToast(data.message || 'Ayrıştırma başarısız oldu', 'error');
        }
    } catch (e) {
        showToast('Sunucu bağlantı hatası', 'error');
    } finally {
        btn.disabled = false;
        btn.innerHTML = origText;
    }
}

function renderAiArchitectTree() {
    const container = document.getElementById('aiArchEpicsContainer');
    if (!container) return;
    container.innerHTML = '';

    const teamUsers = window.ALL_TEAM_USERS || [{ id: 1, name: 'Yönetici' }];
    let totalSubs = 0;
    let totalAiSubs = 0;

    aiArchTreeData.forEach((epic, eIdx) => {
        const epicBox = document.createElement('div');
        epicBox.style.cssText = 'background:var(--bg-card); border:1px solid var(--border-subtle); border-radius:10px; padding:14px; transition:all 0.2s;';
        
        epicBox.ondragover = (ev) => {
            ev.preventDefault();
            epicBox.style.borderColor = '#7B68EE';
            epicBox.style.boxShadow = '0 0 0 2px rgba(123,104,238,0.25)';
        };
        epicBox.ondragleave = () => {
            epicBox.style.borderColor = 'var(--border-subtle)';
            epicBox.style.boxShadow = 'none';
        };
        epicBox.ondrop = (ev) => {
            ev.preventDefault();
            epicBox.style.borderColor = 'var(--border-subtle)';
            epicBox.style.boxShadow = 'none';
            if (!draggedSubtaskRef) return;
            const { epicIdx: fromE, subIdx: fromS } = draggedSubtaskRef;
            if (fromE === eIdx) return;

            const movedItem = aiArchTreeData[fromE].subtasks.splice(fromS, 1)[0];
            if (movedItem) {
                aiArchTreeData[eIdx].subtasks.push(movedItem);
                draggedSubtaskRef = null;
                renderAiArchitectTree();
                showToast('Alt görev yeni modüle taşındı!', 'info');
            }
        };

        const isGated = (epic.execution_mode === 'gated');

        let subtasksHtml = '';
        (epic.subtasks || []).forEach((sub, sIdx) => {
            totalSubs++;
            if (sub.is_ai) totalAiSubs++;

            const userOptions = teamUsers.map(u => `<option value="${u.id}" ${sub.assigned_to == u.id ? 'selected' : ''}>👤 ${escapeHtml(u.name)}</option>`).join('');
            const showDesc = sub.show_desc !== false;

            subtasksHtml += `
                <div draggable="true"
                     ondragstart="draggedSubtaskRef = { epicIdx: ${eIdx}, subIdx: ${sIdx} }; this.style.opacity='0.4';"
                     ondragend="this.style.opacity='1'; draggedSubtaskRef = null;"
                     style="background:var(--bg-base); border:1px solid ${sub.is_ai ? 'rgba(224,86,253,0.5)' : 'var(--border-subtle)'}; border-radius:8px; padding:10px; margin-bottom:8px;">
                    <div style="display:grid; grid-template-columns: 22px 1fr 110px 135px 80px 85px 28px; gap:8px; align-items:center;">
                        <span style="color:var(--text-muted); font-size:15px; cursor:grab; user-select:none;" title="Sürükle ve başka modüle bırak">⠿</span>
                        <input type="text" value="${escapeHtml(sub.title)}" oninput="aiArchTreeData[${eIdx}].subtasks[${sIdx}].title = this.value" placeholder="Alt görev başlığı..." style="background:transparent; border:none; color:var(--text-white); font-size:13px; font-weight:700; outline:none; width:100%;">
                        <select onchange="aiArchTreeData[${eIdx}].subtasks[${sIdx}].category = this.value" style="background:var(--bg-card); border:1px solid var(--border-subtle); color:var(--text-secondary); font-size:11px; padding:4px 6px; border-radius:4px;">
                            <option value="coding" ${sub.category === 'coding' ? 'selected' : ''}>💻 Yazılım</option>
                            <option value="design" ${sub.category === 'design' ? 'selected' : ''}>🎨 Tasarım</option>
                            <option value="seo" ${sub.category === 'seo' ? 'selected' : ''}>🔍 SEO</option>
                            <option value="marketing" ${sub.category === 'marketing' ? 'selected' : ''}>📢 Pazarlama</option>
                            <option value="legal" ${sub.category === 'legal' ? 'selected' : ''}>⚖️ Yasal/KVKK</option>
                            <option value="content" ${sub.category === 'content' ? 'selected' : ''}>📝 İçerik</option>
                        </select>
                        <select onchange="aiArchTreeData[${eIdx}].subtasks[${sIdx}].assigned_to = parseInt(this.value)" style="background:var(--bg-card); border:1px solid var(--border-subtle); color:var(--text-secondary); font-size:11px; padding:4px 6px; border-radius:4px;">
                            ${userOptions}
                        </select>
                        <button type="button" onclick="aiArchTreeData[${eIdx}].subtasks[${sIdx}].show_desc = !aiArchTreeData[${eIdx}].subtasks[${sIdx}].show_desc; renderAiArchitectTree();" style="background:rgba(123,104,238,0.12); border:1px solid rgba(123,104,238,0.3); color:#c4b5fd; font-size:10.5px; font-weight:600; padding:4px 6px; border-radius:4px; cursor:pointer;" title="Nasıl Yapılacak Açıklamasını Göster/Gizle">
                            📝 ${showDesc ? 'Gizle' : 'Detay'}
                        </button>
                        <label style="display:inline-flex; align-items:center; justify-content:center; gap:4px; font-size:11px; font-weight:700; padding:4px 6px; border-radius:5px; cursor:pointer; background:${sub.is_ai ? 'linear-gradient(135deg, #7B68EE, #e056fd)' : 'rgba(255,255,255,0.05)'}; color:${sub.is_ai ? '#fff' : 'var(--text-muted)'}; border:1px solid ${sub.is_ai ? '#e056fd' : 'var(--border-subtle)'};">
                            <input type="checkbox" ${sub.is_ai ? 'checked' : ''} onchange="aiArchTreeData[${eIdx}].subtasks[${sIdx}].is_ai = this.checked; renderAiArchitectTree();" style="display:none;">
                            <span>🤖 AI</span>
                        </label>
                        <button type="button" onclick="aiArchTreeData[${eIdx}].subtasks.splice(${sIdx}, 1); renderAiArchitectTree();" style="background:none; border:none; color:#ff4757; cursor:pointer; font-size:13px;" title="Alt Görevi Sil">🗑️</button>
                    </div>
                    <!-- NASIL YAPILACAK / TEKNİK UYGULAMA AÇIKLAMASI ALANI -->
                    <div style="display:${showDesc ? 'block' : 'none'}; margin-top:8px; padding-top:8px; border-top:1px dashed rgba(255,255,255,0.08); padding-left:30px;">
                        <div style="font-size:10.5px; font-weight:700; color:#a4b0be; margin-bottom:4px;">🛠️ Nasıl Yapılacak? (Uygulama Adımları, Teknik Detaylar & Test Kriterleri):</div>
                        <textarea rows="3" oninput="aiArchTreeData[${eIdx}].subtasks[${sIdx}].description = this.value" placeholder="Bu alt görevin nasıl yapılacağını, teknik gereksinimlerini ve kabul kriterlerini buraya yazın..." style="width:100%; background:rgba(0,0,0,0.25); border:1px solid var(--border-subtle); border-radius:6px; color:var(--text-secondary); font-size:12px; line-height:1.45; padding:8px; resize:vertical; box-sizing:border-box;">${escapeHtml(sub.description || '')}</textarea>
                    </div>
                </div>
            `;
        });

        epicBox.innerHTML = `
            <div style="display:flex; justify-content:space-between; align-items:center; gap:10px; margin-bottom:8px;">
                <div style="display:flex; align-items:center; gap:8px; flex:1;">
                    <span style="font-size:16px;">📂</span>
                    <input type="text" value="${escapeHtml(epic.title)}" oninput="aiArchTreeData[${eIdx}].title = this.value" style="background:transparent; border:1px dashed transparent; color:var(--text-white); font-size:14px; font-weight:800; width:100%; padding:2px 6px; border-radius:4px;" onfocus="this.style.borderColor='var(--brand-primary)'" onblur="this.style.borderColor='transparent'">
                </div>
                <div style="display:flex; align-items:center; gap:8px;">
                    <select onchange="aiArchTreeData[${eIdx}].execution_mode = this.value; renderAiArchitectTree();" style="background:${isGated ? 'rgba(230,126,34,0.18)' : 'rgba(0,184,148,0.18)'}; border:1px solid ${isGated ? '#e67e22' : '#00b894'}; color:${isGated ? '#e67e22' : '#00b894'}; font-size:11px; font-weight:700; padding:4px 8px; border-radius:6px; cursor:pointer;">
                        <option value="auto" ${!isGated ? 'selected' : ''}>🚀 Tam Otonom Akış (Gece Modu)</option>
                        <option value="gated" ${isGated ? 'selected' : ''}>🛑 Onay Kapılı (Adım Adım Onaylı)</option>
                    </select>
                    <button type="button" onclick="addSubtaskToEpic(${eIdx})" style="background:rgba(123,104,238,0.15); border:1px solid rgba(123,104,238,0.4); color:#d8b4fe; font-size:11px; font-weight:700; padding:4px 8px; border-radius:5px; cursor:pointer;">+ Alt Görev</button>
                    <button type="button" onclick="aiArchTreeData.splice(${eIdx}, 1); renderAiArchitectTree();" style="background:rgba(255,71,87,0.12); border:1px solid rgba(255,71,87,0.3); color:#ff4757; font-size:11px; padding:4px 8px; border-radius:5px; cursor:pointer;" title="Modülü Sil">✕</button>
                </div>
            </div>
            <!-- ANA MODÜL AÇIKLAMASI -->
            <div style="margin-bottom:10px; padding-bottom:8px; border-bottom:1px solid var(--border-subtle);">
                <input type="text" value="${escapeHtml(epic.description || '')}" oninput="aiArchTreeData[${eIdx}].description = this.value" placeholder="Ana modülün genel amacı ve kapsam açıklaması..." style="width:100%; background:rgba(0,0,0,0.18); border:1px solid var(--border-subtle); border-radius:5px; color:var(--text-muted); font-size:11.5px; padding:5px 8px; box-sizing:border-box;">
            </div>
            <div>${subtasksHtml || '<div style="font-size:12px; color:var(--text-muted); padding:8px; text-align:center; border:1px dashed var(--border-subtle); border-radius:6px;">Bu modülde alt görev yok. Buraya alt görev sürükleyebilir veya "+ Alt Görev" butonuna basabilirsiniz.</div>'}</div>
        `;
        container.appendChild(epicBox);
    });

    const imgNote = aiArchExtractedImgs.length > 0 ? ` • 🖼️ <b>${aiArchExtractedImgs.length} Döküman Görseli</b> eklenecek` : '';
    document.getElementById('aiArchSummaryBadge').innerHTML = `📊 Toplam <b>${aiArchTreeData.length} Ana Modül</b>, <b>${totalSubs} Detaylı Alt Görev</b> (<b>${totalAiSubs} tanesi 🤖 AI Otonom</b>)${imgNote}`;
}

async function submitAiArchitectTree() {
    if (aiArchTreeData.length === 0) {
        showToast('Kaydedilecek modül bulunamadı.', 'error');
        return;
    }

    const projId = document.getElementById('aiArchProjectSelect').value;
    const btn = document.getElementById('btnApproveAiArch');
    const origText = btn.innerHTML;
    btn.disabled = true;
    btn.innerHTML = '⏳ Projeye İşleniyor...';

    try {
        const fd = new FormData();
        fd.append('project_id', projId);
        fd.append('tree_json', JSON.stringify(aiArchTreeData));
        fd.append('extracted_imgs_json', JSON.stringify(aiArchExtractedImgs));
        fd.append('original_doc_json', JSON.stringify(aiArchOriginalDoc));

        const res = await fetch('/taskmanager/ajax?action=bulk_create_hierarchical_tasks', { method: 'POST', body: fd });
        const data = await res.json();

        if (data.success) {
            showToast(data.message, 'success');
            closeModal('modalAiArchitect');
            setTimeout(() => window.location.reload(), 800);
        } else {
            showToast(data.message || 'Kaydedilemedi', 'error');
        }
    } catch (e) {
        showToast('Sunucu bağlantı hatası', 'error');
    } finally {
        btn.disabled = false;
        btn.innerHTML = origText;
    }
}
