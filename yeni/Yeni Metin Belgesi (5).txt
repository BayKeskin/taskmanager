def is_conversation_active(cid, ls_address, csrf_token=None):
    """
    Sohbetin sadece diskte (.db olarak) durup durmadığına DEĞİL,
    gerçekten Antigravity'de silinmemiş ve aktif olup olmadığına bakar.
    """
    if not cid or not ls_address or not os.path.exists(AGENTAPI_EXE):
        return False

    db_path = os.path.join(ANTIGRAVITY_CONV_DIR, f"{cid}.db")
    if not os.path.exists(db_path):
        return False

    env = os.environ.copy()
    env["ANTIGRAVITY_LS_ADDRESS"] = ls_address
    if csrf_token:
        env["ANTIGRAVITY_CSRF_TOKEN"] = csrf_token

    try:
        # 1. list-conversations varsa oradan kontrol et (Silinen sohbetler listeden düşer!)
        list_res = run_silent(
            [AGENTAPI_EXE, "agentapi", "list-conversations"],
            env=env, capture_output=True, text=True, timeout=3
        )
        if list_res.returncode == 0 and list_res.stdout and "conversationId" in list_res.stdout:
            if cid not in list_res.stdout:
                print(f"🗑️ [SİLİNMİŞ SOHBET TESPİT EDİLDİ] {cid} artık aktif listede yok!")
                return False

        # 2. get-conversation-metadata çıktısını kontrol et
        res = run_silent(
            [AGENTAPI_EXE, "agentapi", "get-conversation-metadata", cid],
            env=env, capture_output=True, text=True, timeout=3
        )
        if res.returncode != 0:
            return False

        out_str = (res.stdout or "").strip()
        if not out_str or "error" in out_str.lower() or "not found" in out_str.lower():
            return False

        # JSON içinde silindiğine dair bayrak (deleted / archived) var mı kontrol et
        try:
            meta_json = json.loads(out_str)
            conv_meta = meta_json.get("response", {}).get("conversationMetadata", meta_json)
            if conv_meta.get("isDeleted") or conv_meta.get("deleted") or conv_meta.get("archived"):
                return False
        except Exception:
            pass

        return True
    except Exception:
        return False


def discover_latest_conversation(project_code, project_title="", workspace_path="", incoming_conv_id=None):
    ls_address, csrf_token = get_antigravity_ls_env()
    workspaces_map = load_workspaces_map()
    target_paths = []
    if workspace_path:
        target_paths.append(workspace_path)
    if project_code in workspaces_map:
        target_paths.extend(workspaces_map[project_code])

    normalized_targets = list(dict.fromkeys([p.replace('\\', '/').lower().rstrip('/') for p in target_paths if p]))
    normalized_targets.sort(key=len, reverse=True)

    if not os.path.exists(ANTIGRAVITY_CONV_DIR):
        return None, None

    db_files = glob.glob(os.path.join(ANTIGRAVITY_CONV_DIR, '*.db'))
    if not db_files:
        return None, None

    db_files.sort(key=os.path.getmtime, reverse=True)

    for db_path in db_files:
        cid = os.path.splitext(os.path.basename(db_path))[0]
        try:
            con = sqlite3.connect(f'file:{db_path}?mode=ro', uri=True)
            cur = con.cursor()
            cur.execute("SELECT data FROM trajectory_metadata_blob WHERE id='main'")
            row = cur.fetchone()
            con.close()
            if row and row[0]:
                decoded_text = urllib.parse.unquote(row[0].decode('utf-8', errors='ignore')).replace('\\', '/').lower()
                for target in normalized_targets:
                    pattern = rf"(?:^|[^\w\-\.]){re.escape(target)}(?:[/?#\"\'\s,\x00-\x1f\\]|$)"
                    if re.search(pattern, decoded_text):
                        if ls_address and not is_conversation_active(cid, ls_address, csrf_token):
                            # Eğer bu sohbet silinmişse diskteki hayalet .db dosyasını da yeniden adlandır/temizle ki bir daha asla takılmasın!
                            try:
                                os.rename(db_path, db_path + ".deleted")
                            except Exception:
                                pass
                            continue
                        return cid, f"file:///{target}"
        except Exception:
            continue

    return None, None
